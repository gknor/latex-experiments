% \myfancybreak{<sign>}
%
% The distances are from baseline to baseline, in units of \baselineskip.
\newcommand*{\fbabove}{2.25}  % last line of the text before -> sign
\newcommand*{\fbbelow}{1.75}  % sign -> first line of the text after
\newcommand*{\fbflex}{0.25}   % stretch and shrink, as a part of the distance
\newcommand*{\fbdepth}{\vphantom{p}} % invisible: the depth of the sign line

\newskip\fb@skip
\gdef\fb@lastbelow{0pt}       % the glue that the last break put below the sign

\newcommand{\myfancybreak}[1]{%
  \par
  \begingroup
    % The glue below the sign. The next paragraph adds \parskip, thus subtract it.
    \fb@skip=\dimexpr\fbbelow\baselineskip-\baselineskip-\parskip\relax
      plus  \fbflex\dimexpr\fbbelow\baselineskip\relax
      minus \fbflex\dimexpr\fbbelow\baselineskip\relax
    \xdef\fb@lastbelow{\the\fb@skip}%
    \ifnum\lastpenalty=\fb@floatmark\relax
      % Directly after a float placed "here". The float already added
      % \intextsep, the same space that a text paragraph gets. Add nothing.
      % The float also added -\parskip for the next paragraph. Cancel it,
      % because the line of the sign has no \parskip.
      \vskip\parskip
    \else
      % Replace the space that a list, a quotation, etc. added below itself.
      % The result is then the same after each type of text.
      \ifdim\lastskip=\z@\else \vskip-\lastskip \fi
      \fb@skip=\dimexpr\fbabove\baselineskip-\baselineskip\relax
        plus  \fbflex\dimexpr\fbabove\baselineskip\relax
        minus \fbflex\dimexpr\fbabove\baselineskip\relax
      \vskip\fb@skip           % plain glue: TeX discards it at the top of a page
    \fi
    % The sign is a normal centered line of text without \parskip. TeX puts
    % it on the baseline grid, as it does for each line of a paragraph.
    % \fbdepth gives the line the depth of a text line with a descender,
    % thus the distance to a float below is the same as for normal text.
    \parskip\z@skip \parindent\z@ \centering
    \noindent #1\fbdepth\par
    \nobreak
  \endgroup
  \vskip\fb@lastbelow\relax
  % A mark after the glue. It hides the glue from \addvspace (a list start
  % would replace the glue), and the float code can find the break.
  \penalty\fb@breakmark\relax
  % Behave as a heading: no page break between the sign and the next text,
  % and a list or a quotation that follows adds no space above itself.
  \@afterindenttrue           % use \@afterindentfalse for no indent after the sign
  \@afterheading
}

% Two changes in the LaTeX code that puts a float "here" (\@addtocurcol).
%
% 1. Before the float. The float adds \intextsep above itself with \vskip, thus
%    this space would add to the space below the sign. If the mark of
%    \myfancybreak is the last item, cancel the glue below the sign first.
\newcommand*{\fb@beforefloat}{%
  \ifnum\lastpenalty=\fb@breakmark\relax
    \fb@skip=\fb@lastbelow\relax \vskip-\fb@skip
  \fi}
\patchcmd{\@addtocurcol}
  {\@cons \@midlist \@currbox}
  {\@cons \@midlist \@currbox \fb@beforefloat}
  {}{\ClassWarning{asterisk-break}{Cannot patch \string\@addtocurcol\space (1)}}
% 2. After the float. If LaTeX put the float "here", put a mark after it, thus
%    \myfancybreak can see that a float is directly before it.
%
% The marks are penalties. TeX reads each value of 10000 or more as "no break",
% but \lastpenalty gives the exact value. A page break after a float is still
% possible at the penalty that LaTeX puts directly after the float box.
\newcommand*{\fb@floatmark}{10001}
\newcommand*{\fb@breakmark}{10002}
\newif\iffb@here
\patchcmd{\@addtocurcol}
  {\outputpenalty \z@ \@inserttrue}
  {\global\fb@heretrue \outputpenalty \z@ \@inserttrue}
  {}{\ClassWarning{asterisk-break}{Cannot patch \string\@addtocurcol\space (2)}}
\pretocmd{\end@float}{\global\fb@herefalse}
  {}{\ClassWarning{asterisk-break}{Cannot patch \string\end@float\space (1)}}
\apptocmd{\end@float}{\ifvmode\iffb@here \penalty\fb@floatmark\relax \fi\fi}
  {}{\ClassWarning{asterisk-break}{Cannot patch \string\end@float\space (2)}}
\makeatother

% --- Tools for this example only ---
\usepackage{zref-savepos}  % saves positions on the page

% \parskip of memoir has a small stretch. Remove it, thus the spaces of the
% sign are the only spaces on the page that can change.
\setlength{\parskip}{0pt}

% \savepos{name} saves the position of a baseline.
\newcommand{\savepos}[1]{\zsaveposy{#1}}
% \distance{a}{b} prints the distance between two marks in units of \baselineskip.
\newcommand{\distance}[2]{\fpeval{round((\zposy{#1}-\zposy{#2})/\number\dimexpr\baselineskip\relax,3)}}
% \measured{n} prints the two distances of break n.
\newcommand{\measured}[1]{above \distance{a#1}{s#1}, below \distance{s#1}{b#1}}
% \lines{m}{n} prints the single lines m to n.
\newcounter{ln}
\newcommand{\lines}[2]{\setcounter{ln}{#1}\addtocounter{ln}{-1}%
  \loop\ifnum\value{ln}<#2 \stepcounter{ln}Line \theln.\par\repeat}
% A block of three lines that TeX cannot divide between two pages.
\newcommand{\block}{{\interlinepenalty=10000
  \noindent A block of three lines. TeX cannot divide it.\\
  This is the second line of the block.\\
  This is the third line of the block.\par}}

\begin{document}
\pagestyle{empty}
% The page has room for 20 lines. A break uses the space of 4 lines:
% 2.25 above + 1.75 below.

% Page 1: 20 lines. The page is full, and the spaces are natural.
\noindent\textbf{Page 1: natural spaces (2.25 and 1.75).}\par
\lines{2}{7}
Line 8, the last line before the sign.\savepos{a1}
\myfancybreak{$\ast$\savepos{s1}}
Line\savepos{b1} 12. Measured: \measured{1}.\par
\lines{13}{20}

% Page 2: 19 lines, then a block of three lines that does not fit.
% TeX stretches the spaces by one line to fill the page.
\noindent\textbf{Page 2: stretched spaces (25\% more).}\par
\lines{2}{7}
Line 8, the last line before the sign.\savepos{a2}
\myfancybreak{$\ast$\savepos{s2}}
Line\savepos{b2} 12 or lower. Measured: \measured{2}.\par
\lines{13}{19}
\block

% Page 3: 21 lines, and the last three lines are a block.
% TeX shrinks the spaces by one line to keep the block on the page.
\noindent\textbf{Page 3: shrunk spaces (25\% less).}\par
\lines{5}{6}
Line 7, the last line before the sign.\savepos{a3}
\myfancybreak{$\ast$\savepos{s3}}
Line\savepos{b3} 11 or higher. Measured: \measured{3}.\par
\lines{12}{18}
\block

\noindent\textbf{Page 4: the end.}\par
The values are in units of \verb|\baselineskip|.
\end{document}
